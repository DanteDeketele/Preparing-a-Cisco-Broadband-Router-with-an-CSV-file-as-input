! Cisco broadband router configuration
hostname Router

! NAT overload for LAN clients using the WAN interface
access-list 1 remark LAN networks allowed to use the WAN
access-list 1 permit 192.168.0.0 0.0.0.255
ip nat inside source list 1 interface gi0/0 overload

ip dhcp excluded-address 192.168.0.1
ip dhcp pool lan
 network 192.168.0.0 255.255.255.0
 default-router 192.168.0.1
 exit

interface gi0/0
 ip nat outside
 description WAN
 ip address 172.23.80.200 255.255.254.0
 no shutdown
exit
ip route 0.0.0.0 0.0.0.0 172.23.80.1

interface gi0/1
 ip nat inside
 description LAN
 ip address 192.168.0.1 255.255.255.0
 no shutdown
exit
